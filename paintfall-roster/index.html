Paintfall Roster

The eleven player frames and the four things that live in the arena tunnels, with their stats and the source for their models.

Player frames

Bars are relative to the best frame in each stat. Armor is hits you can take, jump is a multiple of the base jump, and target size is how big a target you are (smaller is better).

Tunnel enemies

They live on the two tunnel levels, never use the stairs, and only notice you on their own level. A shield blocks all of them.

export function loadLoadout(): Loadout {
  try {
    const saved = { ...DEFAULT_LOADOUT, ...JSON.parse(localStorage.getItem("paintfall-loadout") || "{}") };
    if (saved.paint === "#a3b8ff") saved.paint = PURPLE; // players who picked the old pale purple get the new one
    if (saved.hat !== undefined && !isHatId(saved.hat)) delete saved.hat; // an unknown hat is no hat
    return saved;
  } catch {
    return DEFAULT_LOADOUT;
  }
}

/**
 * Per-character multipliers. fire scales the gun's cooldown, so lower is faster. hit scales how big a target
 * you are for incoming paintballs: a bigger body is easier to hit. Nobody jumps less than 0.8, so every
 * character can still climb the stairs (each step rises 0.6 m).
 */
export type FrameStats = { speed: number; maxHp: number; jump: number; fire: number; hit: number };
export const FRAME_STATS: Record<FrameId, FrameStats> = {
  scout: { speed: 1.25, maxHp: 4, jump: 1, fire: 1, hit: 1 }, // fast and fragile
  tank: { speed: 0.8, maxHp: 8, jump: 0.8, fire: 1.15, hit: 1.15 }, // tough and slow
  orb: { speed: 1, maxHp: 5, jump: 1.5, fire: 0.85, hit: 1.1 }, // the all-rounder with a big jump
  striker: { speed: 1.1, maxHp: 3, jump: 1, fire: 0.7, hit: 0.9 }, // a glass cannon: fastest fire, least armor, small target

/** Painted textures for the sentry, bomber and brute (made once, in the page). */
function enemyTextures() {
  const make = (draw: (g: CanvasRenderingContext2D) => void) => {
    const c = document.createElement("canvas");
    c.width = c.height = 128;
    draw(c.getContext("2d")!);
    const t = new THREE.CanvasTexture(c);
    t.wrapS = t.wrapT = THREE.RepeatWrapping;
    t.colorSpace = THREE.SRGBColorSpace;
    return t;
  };
  // pseudo-random but fixed, so the textures look the same every time
  let seed = 7; const rnd = () => ((seed = (seed * 16807) % 2147483647) / 2147483647);
  return {
    // sentry: yellow and black warning stripes with a worn edge
    hazard: make((g) => {
      g.fillStyle = "#1c1f26"; g.fillRect(0, 0, 128, 128);
      g.fillStyle = "#e8c21a";
      for (let x = -128; x < 256; x += 32) { g.beginPath(); g.moveTo(x, 128); g.lineTo(x + 16, 128); g.lineTo(x + 144, 0); g.lineTo(x + 128, 0); g.closePath(); g.fill(); }
      g.strokeStyle = "#0e1014"; g.lineWidth = 6; g.strokeRect(0, 0, 128, 128);
      g.fillStyle = "rgba(0,0,0,0.25)"; for (let i = 0; i < 40; i++) g.fillRect(rnd() * 128, rnd() * 128, 2 + rnd() * 5, 1);
    }),
    // bomber: a charred shell split by glowing magma cracks
    magma: make((g) => {
      g.fillStyle = "#1d1614"; g.fillRect(0, 0, 128, 128);
      for (let i = 0; i < 260; i++) { g.fillStyle = `rgba(${40 + rnd() * 30},${25 + rnd() * 15},${20},0.6)`; g.fillRect(rnd() * 128, rnd() * 128, 2 + rnd() * 6, 2 + rnd() * 6); }
      g.lineCap = "round";
      for (let k = 0; k < 7; k++) {
        let x = rnd() * 128, y = rnd() * 128;
        g.beginPath(); g.moveTo(x, y);
        for (let n = 0; n < 6; n++) { x += (rnd() - 0.5) * 40; y += (rnd() - 0.5) * 40; g.lineTo(x, y); }
        g.strokeStyle = "#ff7a1a"; g.lineWidth = 3; g.stroke();
        g.strokeStyle = "#ffd23f"; g.lineWidth = 1; g.stroke();
      }
    }),
    // brute: rusted armour plate with seams and rivets
    rust: make((g) => {
      g.fillStyle = "#5b3a2a"; g.fillRect(0, 0, 128, 128);
      for (let i = 0; i < 500; i++) { const r = rnd(); g.fillStyle = r < 0.5 ? "rgba(120,62,30,0.45)" : "rgba(30,18,12,0.4)"; g.fillRect(rnd() * 128, rnd() * 128, 1 + rnd() * 5, 1 + rnd() * 5); }
      g.strokeStyle = "#241510"; g.lineWidth = 3; g.strokeRect(2, 2, 124, 124); g.beginPath(); g.moveTo(64, 0); g.lineTo(64, 128); g.stroke();
      g.fillStyle = "#b08a6a";
      for (const [x, y] of [[12, 12], [116, 12], [12, 116], [116, 116], [52, 64], [76, 64]] as const) { g.beginPath(); g.arc(x, y, 3.5, 0, Math.PI * 2); g.fill(); }
    }),
  };
}

// Crawlers: the enemy that lives in the tunnels. A low, fast, many-legged thing that wanders its own level of the
// tunnels and, when it spots you, rushes you and bites. It never shoots, so the answer is to shoot it first (two hits)
// or keep a corner between you. It stays on its own level (it cannot use the stairs) and only exists in the arena.
// Crawlers are part of your own game (like the arena's bots), so they need no network messages.
import { CELL, TUNNEL_FLOOR, TUNNEL_FLOOR2 } from "./tunnels";
import { bumpStat } from "./ctf";
import { BALL_SPEED, GRAVITY, collide, fire, lineOfSight, type Box, type Events, type V3, type World } from "./sim";

export type EnemyKind = "crawler" | "sentry" | "bomber" | "brute";
export type Crawler = { id: number; kind: EnemyKind; p: V3; hp: number; dead: number; yaw: number; cd: number; target: V3; /** which level it lives on: 1 upper, 2 lower */ level: 1 | 2; /** it saw you (rushing, or aiming) */ chasing: boolean; /** seconds left of an attack that is landing (for the animation) */ bite: number };

/**
 * The four things that live in the tunnels.
 * crawler: the fast biter. sentry: a turret that never moves but shoots from far away. bomber: a rolling mine that races at you and blows up.
 * brute: a big slow armoured walker that is hard to kill and hits hard.
 */
export const ENEMIES: Record<EnemyKind, { name: string; hp: number; r: number; speed: number; rush: number; sight: number; reach: number; cd: number; dmg: number; reward: number; perLevel: number }> = {
  crawler: { name: "Crawler", hp: 2, r: 0.7, speed: 3.2, rush: 7.2, sight: 22, reach: 1.5, cd: 1.1, dmg: 1, reward: 250, perLevel: 4 },
  sentry: { name: "Sentry", hp: 3, r: 0.6, speed: 0, rush: 0, sight: 32, reach: 32, cd: 1.5, dmg: 1, reward: 300, perLevel: 2 },
  bomber: { name: "Bomber", hp: 1, r: 0.55, speed: 4, rush: 10.5, sight: 20, reach: 2.2, cd: 0, dmg: 2, reward: 150, perLevel: 2 },
  brute: { name: "Brute", hp: 6, r: 1.05, speed: 2, rush: 4.6, sight: 22, reach: 2.6, cd: 1.7, dmg: 2, reward: 500, perLevel: 1 },
};

export const CRAWLER_COUNT = 8; // four on each level
export const ENEMY_COUNT = Object.values(ENEMIES).reduce((n, e) => n + 2 * e.perLevel, 0); // every kind, on both levels
export const CRAWLER_HP = 2;
export const CRAWLER_R = 0.7;
export const CRAWLER_SPEED = 3.2; // wandering
export const CRAWLER_RUSH = 7.2; // rushing you (you run at 9)
export const CRAWLER_SIGHT = 22; // how far it notices you
export const CRAWLER_BITE_RANGE = 1.5;
export const CRAWLER_BITE_CD = 1.1;
export const CRAWLER_REWARD = 250; // shot points for a crawler: more than a bot, because it is in a dark tunnel
export const CRAWLER_RESPAWN = 15;
export const SENTRY_SHOT_OWNER = 1000; // paintballs fired by a sentry carry 1000 + its id, so they are not a bot's or a player's

/** Is there room for a crawler here: not inside a crate, pillar or wall? */
export function spotFree(boxes: Box[], p: V3, r = CRAWLER_R + 0.2): boolean {
  for (const b of boxes) {
    if (b.hole || b.max.y <= p.y + 0.01 || b.min.y >= p.y + 1.2) continue;
    if (p.x > b.min.x - r && p.x < b.max.x + r && p.z > b.min.z - r && p.z < b.max.z + r) return false;
  }
  return true;
}

/** A free place on a level, from `spots` (the middles of its tunnel cells). */
export function pickSpot(spots: V3[], boxes: Box[], level: 1 | 2, rand: () => number = Math.random): V3 {
  const y = level === 1 ? TUNNEL_FLOOR : TUNNEL_FLOOR2;
  const here = spots.filter((s) => Math.abs(s.y - y) < 0.01);
  for (let i = 0; i < 60; i++) {
    const s = here[Math.floor(rand() * here.length)]!;
    const p = { x: s.x + (rand() - 0.5) * (CELL - 2), y, z: s.z + (rand() - 0.5) * (CELL - 2) };
    if (spotFree(boxes, p)) return p;
  }
  return { ...here[0]! };
}

export function makeCrawlers(spots: V3[], boxes: Box[], rand: () => number = Math.random): Crawler[] {
  const out: Crawler[] = [];
  for (const kind of Object.keys(ENEMIES) as EnemyKind[]) for (const level of [1, 2] as const) for (let n = 0; n < ENEMIES[kind].perLevel; n++) {
    out.push({ id: out.length, kind, p: pickSpot(spots, boxes, level, rand), hp: ENEMIES[kind].hp, dead: 0, yaw: 0, cd: 1 + rand(), target: pickSpot(spots, boxes, level, rand), level, chasing: false, bite: 0 });
  }
  return out;
}

export function stepCrawlers(w: World, dt: number, ev: Events) {
  const pl = w.player;
  const eye = { x: pl.p.x, y: pl.p.y + 1.5, z: pl.p.z };
  const hurt = (n: number) => {
    if (pl.shield > 0) return;
    pl.hp -= n; ev.playerHit = true;
    if (pl.hp <= 0) { pl.dead = 3; pl.deaths++; bumpStat(w, "me", "deaths"); ev.playerKilled = true; }
  };
  for (const c of w.crawlers) {
    const k = ENEMIES[c.kind];
    if (c.dead > 0) {
      c.dead -= dt;
      if (c.dead <= 0) { c.hp = k.hp; c.p = pickSpot(w.crawlerSpots, w.boxes, c.level); c.target = pickSpot(w.crawlerSpots, w.boxes, c.level); c.chasing = false; c.cd = 1; }
      continue;
    }
    c.bite = Math.max(0, c.bite - dt);
    c.cd = Math.max(0, c.cd - dt);
    const dist = Math.hypot(pl.p.x - c.p.x, pl.p.z - c.p.z);
    const head = { x: c.p.x, y: c.p.y + (c.kind === "sentry" ? 1.2 : c.kind === "brute" ? 1.4 : 0.6), z: c.p.z };
    const onLevel = Math.abs(pl.p.y - c.p.y) < 2.5; // it does not notice you on the other level, or on the ground above
    const sees = pl.dead <= 0 && onLevel && dist < k.sight * (c.chasing && c.kind !== "sentry" ? 1.4 : 1) && lineOfSight(head, eye, w.boxes);
    c.chasing = sees;
    if (c.kind === "sentry") { // never moves: turns to face you and fires, leading its shot like the arena bots
      if (sees) {
        c.yaw = Math.atan2(pl.p.x - c.p.x, pl.p.z - c.p.z);
        if (c.cd <= 0) {
          c.cd = k.cd; c.bite = 0.2;
          const lead = dist / BALL_SPEED, drop = 0.5 * GRAVITY * lead * lead;
          fire(w, head, { x: eye.x - head.x, y: eye.y - 0.3 - head.y + drop, z: eye.z - head.z }, SENTRY_SHOT_OWNER + c.id, 0.05);
        }
      }
      continue;
    }
    let tx = c.target.x - c.p.x, tz = c.target.z - c.p.z;
    if (sees) { tx = pl.p.x - c.p.x; tz = pl.p.z - c.p.z; }
    const tl = Math.hypot(tx, tz);
    if (sees && dist <= k.reach) {
      c.yaw = Math.atan2(tx, tz);
      if (c.kind === "bomber") { // it blows up on you: it hurts, and is gone (no reward: you did not shoot it)
        hurt(k.dmg); ev.crawlerBite = true; c.dead = CRAWLER_RESPAWN; c.bite = 0.35;
        continue;
      }
      if (c.cd <= 0) { c.cd = k.cd; c.bite = 0.35; ev.crawlerBite = true; hurt(k.dmg); }
      continue;
    }
    if (!sees && tl < 1.5) { c.target = pickSpot(w.crawlerSpots, w.boxes, c.level); continue; }
    if (tl > 0.01) {
      const ox = c.p.x, oz = c.p.z, sp = (sees ? k.rush : k.speed) * dt;
      c.p.x += (tx / tl) * sp; c.p.z += (tz / tl) * sp;
      collide(c.p, k.r, w.boxes);
      c.yaw = Math.atan2(tx, tz);
      if (Math.hypot(c.p.x - ox, c.p.z - oz) < sp * 0.2 && !sees) c.target = pickSpot(w.crawlerSpots, w.boxes, c.level); // stuck on something: try elsewhere
    }
  }
